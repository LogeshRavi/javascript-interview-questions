
Flatten an Array (without Array.flat)

function flatten(arr) {
  let result = [];
  for (let item of arr) {
    if (Array.isArray(item)) {
      result.push(...flatten(item));
    } else {
      result.push(item);
    }
  }
  return result;
}

console.log(flatten([1,2,3,[4,5,6,[7,8,[10,11]]],9]))
// [1,2,3,4,5,6,7,8,10,11,9]

First Repeating Character

function firstRepeat(str) {
  let set = new Set()
  for(let item of str){
      if(set.has(item)) {
          return item
      }else{
          set.add(item)
      }
  }
  
  return set
}

console.log(firstRepeat("success"))  ; // "c"

Stopwatch Implementation

let timer;
let seconds = 0;

function start() {
  if (!timer) {
    timer = setInterval(() => {
      seconds++;
      console.log(seconds);
    }, 1000);
  }
}

function stop() {
  clearInterval(timer);
  timer = null;
}

function reset() {
  stop();
  seconds = 0;
}

Currying – Infinite Sum


function sum(a){
    let total = a
    
    return function next(b){
        if(b === undefined) return total
        total += b
        return next
    }
}


console.log(sum(10)(20)(30)())
//60
